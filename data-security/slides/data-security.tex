\begin{frame}{``No training'' does not mean ``no storage''}
  \begin{itemize}
    \item Training, retention, logging, human review, and third-party transfer
          are separate questions.
    \item OpenAI Business, Enterprise, Edu, and API data are not used for
          training by default.
    \item Standard API abuse-monitoring logs may retain customer content for
          up to 30 days.
    \item Zero Data Retention -> Business
    \item Connectors and MCP servers introduce separate providers and policies.
  \end{itemize}
  \vfill
  \href{https://developers.openai.com/api/docs/guides/your-data}{\beamergotobutton{OpenAI API data controls}}\quad
  \href{https://openai.com/enterprise-privacy/}{\beamergotobutton{OpenAI enterprise privacy}}
\end{frame}

\begin{frame}{Documents can contain instructions!}
  \begin{itemize}
    \item PDFs, webpages, READMEs, issue text, emails, and CSV cells can contain
          indirect prompt injections.
    \item An agent may interpret hostile text inside a source as an instruction.
    \item Possible results include injections or data change.
    \item Researchers have already hidden instructions in papers to influence
          AI-assisted peer review.
    \item Retrieved content should be treated as untrusted input.
  \end{itemize}
  \vfill
  \href{https://www.nature.com/articles/d41586-025-02172-y}{\beamergotobutton{Nature: hidden prompts}}\quad
  \href{https://research-and-innovation.ec.europa.eu/news/all-research-and-innovation-news/updated-era-living-guidelines-responsible-use-generative-ai-research-2026-05-08_en}{\beamergotobutton{ERA research guidelines}}
\end{frame}

\begin{frame}{The dangerous combination}
  \begin{itemize}
    \item The highest-risk setup combines private data and
          an outbound communication channel.
    \item Minimize the files, databases, and credentials available to the agent. I haven't let an unrestricted run so far, and capabilities are ok!
    \item Start with full approval requirements!
    \item Require approval for data export, external writes, and other
          consequential actions.
  \end{itemize}
  \vfill
  \href{https://www.cyber.gov.au/business-government/secure-design/artificial-intelligence/careful-adoption-of-agentic-ai-services}{\beamergotobutton{Government agent guidance}}\quad
  \href{https://developers.openai.com/api/docs/guides/tools-connectors-mcp}{\beamergotobutton{OpenAI MCP safety}}
\end{frame}

\begin{frame}{LLMs are smart: Pseudonymised does not mean anonymous}
  \begin{itemize}
    \item Removing names rarely eliminates re-identification risk in rich
          microdata.
    \item Dates, geography, occupation, firms, rare outcomes, and linked
          datasets can reconnect records to people.
    \item Pseudonymised records normally remain personal data in EU.
    \item A model is anonymous only when identification and extraction of
          personal data are both very unlikely.
  \end{itemize}
  \vfill
  \href{https://www.edpb.europa.eu/documents/opinion-of-the-board-art-64/opinion-282024-on-certain-data-protection-aspects-related-to_en}{\beamergotobutton{EDPB Opinion 28/2024}}\quad
  \href{https://ec.europa.eu/eurostat/web/microdata}{\beamergotobutton{Eurostat microdata}}
\end{frame}

\begin{frame}{Reproducibility requires artifacts}
  \begin{itemize}
    \item A chat transcript is not an executable research record.
    \item Models, hosted tools, search results, and dependencies can change.
    \item Preserve accepted code, human edits, data hashes, dependency
          lockfiles, and random seeds.
    \item Record the model or product, date, tool configuration, consequential
          prompts, and validation results.
    \item Rerun the main results from a clean environment before publication.
  \end{itemize}
  \vfill

\end{frame}

\begin{frame}{Main risks for empirical research}
  \begin{itemize}
    \item \textbf{Prompt injection (rare if you are careful):} a paper, webpage, or dataset changes the
          agent's instructions.
    \item \textbf{Privileges:} the agent inherits broad Git, database,
          cloud, or filesystem access.
    \item \textbf{Chain of Tools (!!):} a package, skill, plugin, or MCP server is
          malicious or compromised.
    \item \textbf{Research integrity:} generated code introduces a bad merge,
          wrong specification, or fabricated citation. This happens more often than you think!
    \item \textbf{Output leakage:} small cells, logs, residuals, or traces
          disclose protected information.
  \end{itemize}

\begin{frame}{Recommendedations for security}
  \begin{itemize}
\item Keep raw data and environment variablesoutside the agent workspace.
    \item Keep network access off or allowlist exact destinations.
    \item Require approval for installation, export, deletion, publication,
          messaging, and Git pushes!
    \item Use only trusted connectors, skills and MCPs.
    \item Always review the code and the output!
  \end{itemize}
